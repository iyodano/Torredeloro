    <script>
        (function () {
            const clip = document.getElementById('pizzaClip');
            const container = document.getElementById('pizzaContainer');
            const pizzas = container.querySelectorAll('img');
            const nameDisplay = document.getElementById('pizza-name');

            let currentIndex = 0;
            let rotation = 0;
            let animating = false;

            const DURACION = 800;
            const SWAP_EN = 0.5;

            // Precalentamiento para evitar parpadeos
            function precalentarImagenes() {
                const originalActive = currentIndex;
                const originalTransition = container.style.transition;
                container.style.transition = 'none';
                
                for (let i = 0; i < pizzas.length; i++) {
                    if (i !== originalActive) {
                        pizzas[originalActive].classList.remove('active');
                        pizzas[i].classList.add('active');
                        void container.offsetHeight;
                        window.getComputedStyle(pizzas[i]).opacity;
                        setTimeout(() => {
                            pizzas[i].classList.remove('active');
                            pizzas[originalActive].classList.add('active');
                            void container.offsetHeight;
                        }, 30);
                    }
                }
                setTimeout(() => {
                    container.style.transition = originalTransition;
                }, 200);
            }

            window.addEventListener('load', () => {
                setTimeout(precalentarImagenes, 80);
                // Ajuste dinámico de la altura del clip (siempre la mitad exacta del ancho)
                function actualizarAlturaClip() {
                    if (clip) {
                        const anchoReal = clip.clientWidth;
                        clip.style.height = (anchoReal / 2) + 'px';
                    }
                }
                actualizarAlturaClip();
                window.addEventListener('resize', actualizarAlturaClip);
            });

            function girarPizza() {
                if (animating) return;
                animating = true;

                rotation += 360;
                container.style.transition = `transform ${DURACION}ms cubic-bezier(0.65, 0, 0.35, 1)`;
                container.style.transform = `rotate(${rotation}deg)`;

                setTimeout(() => {
                    pizzas[currentIndex].classList.remove('active');
                    currentIndex = (currentIndex + 1) % pizzas.length;
                    pizzas[currentIndex].classList.add('active');
                    nameDisplay.textContent = pizzas[currentIndex].dataset.name;
                }, DURACION * SWAP_EN);

                setTimeout(() => {
                    animating = false;
                    container.style.transition = '';
                }, DURACION + 20);
            }

            // El click se captura en el área recortada (mitad visible de la pizza)
            clip.addEventListener('click', e => {
                e.stopPropagation();
                girarPizza();
            });
            container.addEventListener('dragstart', e => e.preventDefault());

            requestAnimationFrame(() => {
                container.style.transition = 'none';
                container.style.transform = 'rotate(0deg)';
            });
        })();
    </script>
